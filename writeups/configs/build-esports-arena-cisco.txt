! ===== DNS-ALLOW (10.90.20.53) — Unbound: every name is NXDOMAIN unless it's on the list =====
ip access-list extended V10-IN
 remark PLAYERS: match servers, the allowlist resolver, HTTPS out
 10 permit udp 10.90.10.0 0.0.0.255 10.90.15.0 0.0.0.255 range 27015 27020
 20 permit udp 10.90.10.0 0.0.0.255 host 10.90.20.53 eq domain
 30 permit tcp 10.90.10.0 0.0.0.255 host 10.90.20.53 eq domain
 40 permit tcp 10.90.10.0 0.0.0.255 10.90.20.0 0.0.0.255 established
 50 deny   ip any 10.0.0.0 0.255.255.255 log
 60 deny   ip any 172.16.0.0 0.15.255.255
 70 deny   ip any 192.168.0.0 0.0.255.255
 80 permit tcp 10.90.10.0 0.0.0.255 any eq 443
 90 deny   ip any any log
!
ip access-list extended V15-IN
 remark MATCH SERVERS: answer players and observers, report results
 10 permit udp 10.90.15.0 0.0.0.255 range 27015 27020 10.90.10.0 0.0.0.255
 20 permit udp 10.90.15.0 0.0.0.255 eq 27020 10.90.30.0 0.0.0.255
 30 permit tcp 10.90.15.0 0.0.0.255 10.90.20.0 0.0.0.255 established
 40 deny   ip any 10.0.0.0 0.255.255.255 log
 50 permit tcp 10.90.15.0 0.0.0.255 any eq 443
 60 deny   ip any any log
!
ip access-list extended UPLINK-IN
 remark The pod's only door: delayed spectator feed in, replies in, nothing else
 10 permit udp 10.90.30.0 0.0.0.255 10.90.15.0 0.0.0.255 eq 27020
 20 permit tcp any 10.90.10.0 0.0.0.255 established
 30 permit tcp any 10.90.15.0 0.0.0.255 established
 40 permit tcp any 10.90.20.0 0.0.0.255 established
 50 permit udp any eq domain host 10.90.20.53
 60 deny   ip any any log
!

interface Vlan15

! ===== Capture box — 10-minute files, 48 hours kept =====
interface GigabitEthernet0/0/2.900
 encapsulation dot1Q 900
 ip address 10.90.254.1 255.255.255.248
 ip nat inside
 zone-member security INSIDE
 ip policy route-map PBR-IN
interface GigabitEthernet0/0/2.50
 encapsulation dot1Q 50
 ip address 10.90.48.1 255.255.248.0
 ip nat inside
 zone-member security PUBLIC
interface GigabitEthernet0/0/0
 description ISP A
 ip address 203.0.113.10 255.255.255.248
 ip nat outside
 zone-member security OUTSIDE-A
interface GigabitEthernet0/0/1
 description ISP B
 ip address 198.51.100.10 255.255.255.248
 ip nat outside
 zone-member security OUTSIDE-B
!
ip sla 1
 icmp-echo 1.1.1.1 source-interface GigabitEthernet0/0/0
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
 delay down 10 up 60
ip sla 2
 icmp-echo 9.9.9.9 source-interface GigabitEthernet0/0/1
 frequency 5
ip sla schedule 2 life forever start-time now
track 2 ip sla 2 reachability
ip route 1.1.1.1 255.255.255.255 203.0.113.9            ! each probe pinned to its own ISP
ip route 1.1.1.1 255.255.255.255 Null0 250
ip route 9.9.9.9 255.255.255.255 198.51.100.9
ip route 9.9.9.9 255.255.255.255 Null0 250
ip route 0.0.0.0 0.0.0.0 203.0.113.9 track 1
ip route 0.0.0.0 0.0.0.0 198.51.100.9 250
ip route 10.90.0.0 255.255.0.0 10.90.254.2
!
! encoder 1 always leaves via ISP B while ISP B is healthy
ip access-list extended ENC1
 permit ip host 10.90.30.21 any
route-map PBR-IN permit 10
 match ip address ENC1
 set ip next-hop verify-availability 198.51.100.9 1 track 2
!
route-map NAT-A permit 10
 match interface GigabitEthernet0/0/0
route-map NAT-B permit 10
 match interface GigabitEthernet0/0/1
ip nat inside source route-map NAT-A interface GigabitEthernet0/0/0 overload
ip nat inside source route-map NAT-B interface GigabitEthernet0/0/1 overload
! ===== EDGE — who may use which ISP (zone pairs) =====
zone security INSIDE
zone security PUBLIC
zone security OUTSIDE-A
zone security OUTSIDE-B
class-map type inspect match-any CM-ANY
 match protocol tcp
 match protocol udp
 match protocol icmp
policy-map type inspect PM-OUT
 class type inspect CM-ANY
  inspect
 class class-default
  drop log
zone-pair security ZP-IN-A source INSIDE destination OUTSIDE-A
 service-policy type inspect PM-OUT
zone-pair security ZP-IN-B source INSIDE destination OUTSIDE-B
 service-policy type inspect PM-OUT
zone-pair security ZP-PUB-A source PUBLIC destination OUTSIDE-A
 service-policy type inspect PM-OUT
! no PUBLIC → OUTSIDE-B pair: when ISP A fails, spectators simply have no internet
! ===== EDGE — classify on the way in, queue on the way out =====
ip access-list extended CRIT
 permit ip 10.90.10.0 0.0.0.255 any
 permit ip 10.90.15.0 0.0.0.255 any
 permit ip 10.90.20.0 0.0.0.255 any
 permit ip host 10.90.30.22 any
class-map match-any CRIT
 match access-group name CRIT
policy-map MARK-IN
 class CRIT
  set qos-group 1
interface GigabitEthernet0/0/2.900
 service-policy input MARK-IN
!
class-map match-any CRIT-Q
 match qos-group 1
policy-map WAN-A-CHILD
 class CRIT-Q
  bandwidth remaining percent 60
 class class-default
  fair-queue
policy-map WAN-A
 class class-default
  shape average 475000000
  service-policy WAN-A-CHILD
interface GigabitEthernet0/0/0
 service-policy output WAN-A
