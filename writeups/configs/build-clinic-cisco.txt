! ===== R1 — CAPsMAN for three cAP ax, per-device keys on CLINIC-STAFF =====

! each tablet's MAC maps to its own PSK via RADIUS/ISE; unknown MACs are rejected

! ===== R1 — fibre, 5G passthrough, recursive failover, NAT =====
! R1 — fibre primary, 5G floating backup; IP SLA + track, not recursive routes
interface GigabitEthernet0/0/0
 description fibre
 ip address 203.0.113.50 255.255.255.248
 ip nat outside
interface Cellular0/1/0
 description 5G backup
 ip nat outside
ip sla 1
 icmp-echo 1.1.1.1 source-interface GigabitEthernet0/0/0
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
ip route 0.0.0.0 0.0.0.0 203.0.113.49 1 track 1
ip route 0.0.0.0 0.0.0.0 Cellular0/1/0 2
ip access-list standard NAT-ALL
 permit 10.60.0.0 0.0.255.255
ip nat inside source list NAT-ALL interface GigabitEthernet0/0/0 overload
! patients and the archive never ride the metered 5G: PBR pins them to the fibre
ip access-list extended NO-5G
 permit ip 10.60.50.0 0.0.0.255 any
 permit ip 10.60.35.0 0.0.0.255 any
route-map PBR-NO5G permit 10
 match ip address NO-5G
 set interface GigabitEthernet0/0/0
interface Gi0/0/1.50
 ip policy route-map PBR-NO5G
! ===== R1 — mark real-time traffic, queue tree on the fibre =====
! R1 — C8200L: LLQ under a shaper on the fibre
! read top-down, first match wins; wildcard masks inverted. Deny RFC1918, then permit internet
ip access-list extended TELEMED
 permit udp 10.60.10.0 0.0.0.255 any
class-map match-any REALTIME
 match dscp ef
 match access-group name TELEMED
policy-map WAN-CHILD
 class REALTIME
  priority 20000
 class class-default
  fair-queue
policy-map WAN-OUT
 class class-default
  shape average 95000000
  service-policy WAN-CHILD
interface GigabitEthernet0/0/0
 service-policy output WAN-OUT
! ===== R1 — a disabled rule and two schedules =====
! R1 — a time-range on the inbound ACL of the device subinterface.

 25 permit tcp host 10.60.30.11 object-group VENDOR-HUB eq 443 time-range VENDOR-XRAY log
