! ===== wAP 60G pair — a transparent bridge for the trunk =====
! No Cisco 60 GHz product fits a single villa. The same gigabit link to the annex
! is either a buried fibre/copper run, or a PtP bridge from Cisco's Ultra-Reliable
! Wireless Backhaul / Catalyst IW line — far more kit and cost than a pair of
! wAP 60G. For one villa, MikroTik is the right call; the trunk it carries
! (VLANs 10/20/30/99) is identical whatever bridges the 40 m gap.

! ===== R1 — interface lists, input and forward chains =====
! R1 — Zone-Based Firewall. No zone-pair = dropped (default-deny).
zone security OWNER
zone security GUEST
zone security HOME
zone security CCTV
zone security WAN
interface Vlan10
 zone-member security OWNER
interface Vlan20
 zone-member security GUEST
interface Vlan30
 zone-member security HOME
interface Vlan40
 zone-member security CCTV
interface GigabitEthernet0/0/0
 zone-member security WAN
class-map type inspect match-any CM-GENERAL
 match protocol tcp
 match protocol udp
 match protocol icmp
policy-map type inspect PM-OUT
 class type inspect CM-GENERAL
  inspect
 class class-default
  drop
! every inside zone reaches the internet
zone-pair security ZP-OWNER-WAN source OWNER destination WAN
 service-policy type inspect PM-OUT
zone-pair security ZP-GUEST-WAN source GUEST destination WAN
 service-policy type inspect PM-OUT
zone-pair security ZP-HOME-WAN source HOME destination WAN
 service-policy type inspect PM-OUT
zone-pair security ZP-CCTV-WAN source CCTV destination WAN
 service-policy type inspect PM-OUT
! the owner reaches the smart home and the cameras; nobody else crosses
zone-pair security ZP-OWNER-HOME source OWNER destination HOME
 service-policy type inspect PM-OUT
zone-pair security ZP-OWNER-CCTV source OWNER destination CCTV
 service-policy type inspect PM-OUT
! the owner casts to the TVs, which sit in the guest VLAN (.32-.35)
ip access-list extended ACL-TVS
 permit ip any 10.70.20.32 0.0.0.3
class-map type inspect match-all CM-TVS
 match access-group name ACL-TVS
policy-map type inspect PM-TVS
 class type inspect CM-TVS
  inspect
zone-pair security ZP-OWNER-GUEST source OWNER destination GUEST
 service-policy type inspect PM-TVS
! cross-VLAN mDNS for casting = IOS-XE Service Discovery Gateway (mdns-sd gateway)
! ===== R1 — Starlink on ether1, probe route maintained by the DHCP client =====
! R1 — Starlink in bypass: the WAN takes the carrier-NAT address; IP SLA tracks the path
interface GigabitEthernet0/0/0
 description Starlink (bypass)
 ip address dhcp
 ip nat outside
ip sla 1
 icmp-echo 1.1.1.1
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
ip route 0.0.0.0 0.0.0.0 dhcp 1 track 1
! the dish's own status page stays reachable on a static route
ip route 192.168.100.1 255.255.255.255 GigabitEthernet0/0/0
ip access-list standard NAT-ALL
 permit 10.70.0.0 0.0.255.255
ip nat inside source list NAT-ALL interface GigabitEthernet0/0/0 overload
! ===== R1 — 4G as distance 2, shaped for guests, closed to uploads =====
! R1 — ISR 1100 with built-in LTE: the LTE interface is its own zone,
! and only the guest and owner zones get a pair to it (cameras and smart home don't)
zone security LTE
interface Cellular0/2/0
 zone-member security LTE
zone-pair security ZP-GUEST-LTE source GUEST destination LTE
 service-policy type inspect PM-OUT
zone-pair security ZP-OWNER-LTE source OWNER destination LTE
 service-policy type inspect PM-OUT
! guests on LTE: police 10 Mb/s in an output policy on Cellular0/2/0
! ===== R1 — CAKE on the Starlink link =====
! R1 — no CAKE on IOS: a shaper with fair-queue gives per-FLOW fairness,
! not per-host fairness behind NAT. Set it just under typical evening capacity.
policy-map STARLINK-CHILD
 class class-default
  fair-queue
policy-map STARLINK-OUT
 class class-default
  shape average 20000000
  service-policy STARLINK-CHILD
interface GigabitEthernet0/0/0
 service-policy output STARLINK-OUT
! ===== R1 — rotate the guest key and send it to the manager =====
! Cisco has no on-box random key + e-mail. The usual pattern is a small scheduled
! script on the operator's hub that generates the key and pushes one line to the
! wireless controller (over SSH or RESTCONF), then mails the manager:
wlan VILLA-Guest 1 VILLA-Guest
 security wpa akm psk set-key ascii 0 <new-14-character-key>
! changing the key re-keys the WLAN and disconnects the previous guests' devices
! ===== R1 — WireGuard to the operator's hub, alerts and remote logs =====
! R1 — FlexVPN client: the router dials out to the operator's hub through CGNAT
crypto ikev2 profile TO-HUB
 match identity remote fqdn hub.example.net
 identity local fqdn villa-r1.example.net
 authentication local rsa-sig
 authentication remote rsa-sig
 pki trustpoint OP-CA
 dpd 10 3 periodic
crypto ipsec profile TO-HUB
 set ikev2-profile TO-HUB
interface Tunnel1
 ip address 10.250.0.21 255.255.255.0
 tunnel source GigabitEthernet0/0/0
 tunnel destination dynamic
 tunnel protection ipsec profile TO-HUB
crypto ikev2 client flexvpn HUB
 peer 1 fqdn hub.example.net dynamic
 client connect Tunnel1
! alerts + remote logging to the hub; EEM is the netwatch equivalent
logging host 10.250.0.1
logging trap warnings
event manager applet WAN-DOWN
 event track 1 state down
 action 1 syslog msg "Starlink path DOWN"
