! ===== CORE-ADMIN — loopback and ring ports =====
hostname CORE-ADMIN
! the cores route; a ring of routed ports means no spanning tree across the quay
ip routing
interface Loopback0
 description router-id + management anchor
 ip address 10.255.0.1 255.255.255.255
interface TenGigabitEthernet1/1/1
 description RING to CORE-TERM Te1/1/1

interface TenGigabitEthernet1/1/2
 description RING to CORE-TOWER Te1/1/2

! ===== CORE-ADMIN — OSPF process, area 0 on the ring =====
router ospf 1
 router-id 10.255.0.1
! 100G reference: 10G = cost 10, 1G = cost 100 (the default 100M makes them all 1)
 auto-cost reference-bandwidth 100000
! nothing talks OSPF unless named below: no hellos into a user VLAN, ever
 passive-interface default
 no passive-interface TenGigabitEthernet1/1/1
 no passive-interface TenGigabitEthernet1/1/2
 no passive-interface TenGigabitEthernet1/1/3
interface Loopback0
 ip ospf 1 area 0
interface TenGigabitEthernet1/1/1
! two routers on a cable: no DR/BDR election, adjacency comes up faster
 ip ospf network point-to-point
 ip ospf 1 area 0
interface TenGigabitEthernet1/1/2
 ip ospf network point-to-point
 ip ospf 1 area 0
interface TenGigabitEthernet1/1/3
 ip ospf network point-to-point
 ip ospf 1 area 0
! ===== CORE-ADMIN — HMAC-SHA-256 key chain on every adjacency =====
key chain OSPF-KEYS
 key 1
  key-string ********
! RFC 5709: HMAC-SHA-256, not MD5 and never plain text
  cryptographic-algorithm hmac-sha-256
interface TenGigabitEthernet1/1/1
 ip ospf authentication key-chain OSPF-KEYS
interface TenGigabitEthernet1/1/2
 ip ospf authentication key-chain OSPF-KEYS
interface TenGigabitEthernet1/1/3
 ip ospf authentication key-chain OSPF-KEYS
router ospf 1
! drop any OSPF packet that didn't come from a directly connected neighbour
 ttl-security all-interfaces hops 1

! ===== EDGE1 — fibre, tracked default, NAT and default origination =====
hostname EDGE1
interface Loopback0
 ip address 10.255.0.11 255.255.255.255
interface GigabitEthernet0/0/1
 description LAN to CORE-ADMIN Te1/1/4
 ip address 10.255.1.11 255.255.255.254
 ip nat inside
 ip ospf network point-to-point
 ip ospf 1 area 0
 ip ospf authentication key-chain OSPF-KEYS
key chain OSPF-KEYS
 key 1
  key-string ********
  cryptographic-algorithm hmac-sha-256
interface GigabitEthernet0/0/0
 description WAN fibre ISP (static /29)
 ip address 203.0.113.2 255.255.255.248
 ip nat outside
! the probe target is pinned to the ISP so it can't follow OSPF's default
ip route 9.9.9.9 255.255.255.255 203.0.113.1
ip sla 1
 icmp-echo 9.9.9.9 source-interface GigabitEthernet0/0/0
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
 delay down 10 up 30
ip route 0.0.0.0 0.0.0.0 203.0.113.1 track 1
! the building /20s plus anything unknown inside: never leak 10/8 to the ISP
ip route 10.0.0.0 255.0.0.0 Null0
ip access-list extended NAT-OUT
 permit ip 10.0.1.0 0.255.0.255 any
 permit ip 10.0.2.0 0.255.0.255 any
 permit ip 10.20.3.0 0.0.0.255 any
 permit ip 10.30.6.0 0.0.0.255 any
 permit ip 10.10.12.0 0.0.0.255 any
ip nat inside source list NAT-OUT interface GigabitEthernet0/0/0 overload
router ospf 1
 router-id 10.255.0.11
 auto-cost reference-bandwidth 100000
 passive-interface default
 no passive-interface GigabitEthernet0/0/1
 ttl-security all-interfaces hops 1
! only while this edge's own default is installed (no 'always')
 default-information originate metric 10 metric-type 1
interface Loopback0
 ip ospf 1 area 0

! ===== CORE-TERM — the terminal's SVI ACLs (CORE-ADMIN, CARGO, TOWER and YARD use the same ACL text) =====
ip access-list extended STAFF-IN
 remark staff: anywhere useful, never into the security or OT segments
 deny   ip any 10.0.4.0 0.255.0.255
 deny   ip any 10.0.5.0 0.255.0.255
 deny   ip any 10.0.8.0 0.255.0.255
 deny   ip any 10.0.9.0 0.255.0.255
 deny   ip any 10.0.15.0 0.255.0.255
 permit ip any any
ip access-list extended TICKET-IN
 remark ticket desks + boarding gates: DHCP/DNS/NTP inside, booking platform outside
 permit udp any eq bootpc any eq bootps
 permit udp any host 10.10.12.10 eq domain
 permit udp any host 10.10.12.5 eq ntp
 deny   ip any 10.0.0.0 0.255.255.255
 permit tcp any any eq 443
ip access-list extended CCTV-IN
 remark cameras answer the NVR, get time, send logs: nothing else
 permit tcp 10.0.4.0 0.255.0.255 host 10.10.12.30 established
 permit udp any eq bootpc any eq bootps
 permit udp 10.0.4.0 0.255.0.255 host 10.10.12.5 eq ntp
 permit udp 10.0.4.0 0.255.0.255 host 10.10.12.20 eq syslog
ip access-list extended ACCESS-IN
 remark door and gate controllers talk to the access-control server only
 permit tcp 10.0.5.0 0.255.0.255 host 10.10.12.40 eq 443
 permit tcp 10.0.5.0 0.255.0.255 host 10.10.12.40 established
 permit udp any eq bootpc any eq bootps
 permit udp 10.0.5.0 0.255.0.255 host 10.10.12.5 eq ntp

! ===== CORE-TOWER, CORE-CARGO, YARD — the segment-specific ACLs =====
zone security INSIDE
zone security OUTSIDE
class-map type inspect match-any INSIDE-TO-INET
 match access-group name NAT-OUT
policy-map type inspect INSIDE-TO-INET
 class type inspect INSIDE-TO-INET
  inspect
 class class-default
  drop log
zone-pair security IN-OUT source INSIDE destination OUTSIDE
 service-policy type inspect INSIDE-TO-INET
! the router itself: replies to its own probes and DHCP from the dish
ip access-list extended WAN-TO-SELF
 permit udp any eq bootps any eq bootpc
 permit icmp any any echo-reply
 permit icmp any any unreachable
class-map type inspect match-any WAN-TO-SELF
 match access-group name WAN-TO-SELF
policy-map type inspect WAN-TO-SELF
 class type inspect WAN-TO-SELF
  pass
 class class-default
  drop log
zone-pair security OUT-SELF source OUTSIDE destination self
 service-policy type inspect WAN-TO-SELF
interface GigabitEthernet0/0/1

interface GigabitEthernet0/0/0
