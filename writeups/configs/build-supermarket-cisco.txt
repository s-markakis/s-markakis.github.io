! ===== ACC-FRONT — C9200L-24P-4G · VLANs and lane ports =====
hostname ACC-FRONT
! keeps its own VLAN list; ignores VTP from the wire (a rogue switch can't wipe it)
vtp mode transparent
! fast (802.1w) loop prevention, one instance per VLAN
spanning-tree mode rapid-pvst
vlan 10
 name CDE
vlan 20
 name OFFICE
vlan 30
 name HANDHELD
vlan 40
 name STORE-IOT
vlan 45
 name REFRIG
vlan 50
 name GUEST
vlan 60
 name CCTV
vlan 99
 name MGMT
vlan 666
 name PARKING
!
! lanes: one POS and one PIN pad per lane, each pinned to its own MAC
interface range GigabitEthernet1/0/1 - 16
 description LANE POS / PIN PAD
 switchport mode access
 switchport access vlan 10
 ! caps the MACs a port may learn; a violation locks it — no swapping a camera for a laptop
switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 switchport port-security
 spanning-tree portfast
 ! portfast skips the listen/learn wait; bpduguard shuts the port on a switch's BPDU
 spanning-tree bpduguard enable
interface range GigabitEthernet1/0/17 - 20
 description SELF-CHECKOUT
 switchport mode access
 switchport access vlan 10
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security
 spanning-tree portfast
 spanning-tree bpduguard enable
interface GigabitEthernet1/0/21
 description SIGNAGE
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
interface range GigabitEthernet1/0/22 , GigabitEthernet1/0/23 - 24
 switchport access vlan 666
 shutdown                                    ! Gi1/0/23-24 become AP trunks in step 5
!
! a swapped PIN pad stays down until a person looks
no errdisable recovery cause psecure-violation
