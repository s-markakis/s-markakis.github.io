! ===== R1 — payment VLAN rules (first in the forward chain) =====
! R1 — C1121X: IOS ACLs can't match host names, so the acquirer's
! published address ranges go into an object-group
object-group network ACQUIRER
 198.51.100.0 255.255.255.0
! read top-down, first match wins; wildcard masks inverted. Deny RFC1918, then permit internet
ip access-list extended V25-IN
 permit udp 10.40.25.0 0.0.0.255 host 10.40.25.1 eq domain
 permit tcp 10.40.25.0 0.0.0.255 object-group ACQUIRER eq 443
 deny   ip any any log
ip access-list extended V25-OUT
 permit tcp any 10.40.25.0 0.0.0.255 established
 deny   ip any any log                          ! nothing starts a connection into payments
interface Vlan25
 ip access-group V25-IN in
 ip access-group V25-OUT out

interface Vlan30

interface Vlan40
 zone-member security CCTV
interface Vlan50

interface Dialer1

  inspect

  drop

! ===== R1 — hotspot on the guest VLAN, click-through ("trial") login =====

! per-client 5 Mb/s: a QoS policy applied per client on the GUEST policy profile
! ===== R1 — total guest cap with a queue tree (on top of the hotspot's per-device queues) =====
! R1 — one shaper for the whole guest VLAN; per-device fairness is the controller's job
policy-map GUEST-DOWN
 class class-default
  shape average 180000000
  fair-queue
interface Vlan50
 service-policy output GUEST-DOWN
! the 30 Mb/s upload cap is a matching shaper on guest traffic as it egresses the WAN
policy-map GUEST-UP
 class class-default
  shape average 30000000
  fair-queue
! ===== R1 — PPPoE fibre, 4G passthrough, recursive failover, NAT =====
! R1 — PPPoE dialer, built-in LTE, IP SLA instead of recursive routes
interface GigabitEthernet0/0/0
 pppoe-client dial-pool-number 1
interface Dialer1
 ip address negotiated
 encapsulation ppp
 dialer pool 1
 ppp chap hostname cafe@isp
 ppp chap password 0 ********
 ip mtu 1492
 ip tcp adjust-mss 1452
 ip nat outside
ip sla 1
 icmp-echo 1.1.1.1 source-interface Dialer1
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
ip route 1.1.1.1 255.255.255.255 Dialer1
ip route 0.0.0.0 0.0.0.0 Dialer1 track 1
ip route 0.0.0.0 0.0.0.0 Cellular0/1/0 250

crypto ipsec profile TO-HUB-IPSEC

interface Tunnel1
