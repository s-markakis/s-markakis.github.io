! ===== SW1 and SW2 — the OFFICE-NAC policy and the desk ports in open mode =====
class-map type control subscriber match-all DOT1X_NO_RESP
 match method dot1x
 match result-type method dot1x agent-not-found
class-map type control subscriber match-all DOT1X_FAILED
 match method dot1x
 match result-type method dot1x authoritative
class-map type control subscriber match-all MAB_FAILED
 match method mab
 match result-type method mab authoritative
policy-map type control subscriber OFFICE-NAC
 event session-started match-all
  10 class always do-until-failure
   10 authenticate using dot1x priority 10
! entry 10 is left free: step 8 puts 'NPS is down' there, ahead of these
 event authentication-failure match-first
  20 class DOT1X_NO_RESP do-until-failure
   10 terminate dot1x
   20 authenticate using mab priority 20
  30 class DOT1X_FAILED do-until-failure
   10 terminate dot1x
   20 authentication-restart 60
  40 class MAB_FAILED do-until-failure
   10 terminate mab
   20 authentication-restart 60
  50 class always do-until-failure
   10 terminate dot1x
   20 terminate mab
   30 authentication-restart 60
 event agent-found match-all
  10 class always do-until-failure
   10 terminate mab
   20 authenticate using dot1x priority 10

! ===== SW1 and SW2 — critical VLAN, voice, and recovery when NPS returns =====
service-template CRIT-DATA
 vlan 60
class-map type control subscriber match-all AAA_DOWN_UNAUTHD
 match result-type aaa-timeout
 match authorization-status unauthorized
class-map type control subscriber match-all AAA_DOWN_AUTHD
 match result-type aaa-timeout
 match authorization-status authorized
class-map type control subscriber match-any IN_CRIT
 match activated-service-template CRIT-DATA
 match activated-service-template DEFAULT_CRITICAL_VOICE_TEMPLATE
policy-map type control subscriber OFFICE-NAC
 event authentication-failure match-first
! NPS unreachable, new device: CRIT VLAN for data, voice keeps working
  10 class AAA_DOWN_UNAUTHD do-until-failure
   10 activate service-template CRIT-DATA
   20 activate service-template DEFAULT_CRITICAL_VOICE_TEMPLATE
   30 authorize
   40 pause reauthentication
! NPS unreachable at reauth time: keep what it already granted
  15 class AAA_DOWN_AUTHD do-until-failure
   10 pause reauthentication
   20 authorize
 event aaa-available match-all
! NPS is back: throw out the critical sessions so they authenticate properly
  10 class IN_CRIT do-until-failure
   10 clear-session
  20 class always do-until-failure
   10 resume reauthentication
