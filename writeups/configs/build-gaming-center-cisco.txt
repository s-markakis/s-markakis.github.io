! ===== SW-SEATS — bridge, VLANs, seat ports =====
! SW-SEATS — Catalyst 9300-48T: 40 seat ports on VLAN 10, uplink trunk to R1

 switchport mode access

 switchport nonegotiate

 spanning-tree bpduguard enable

interface TenGigabitEthernet1/1/3
 description uplink to R1
 switchport mode trunk
 switchport trunk allowed vlan 10,30,99
interface Vlan99
 ip address 10.30.99.11 255.255.255.0
ip default-gateway 10.30.99.1
! ===== SW-SEATS — port isolation in the switch chip =====
! SW-SEATS — Catalyst 9300-48T + C9300-NM-8X: protected ports are Cisco's port isolation
interface range GigabitEthernet1/0/1 - 40
 description SEAT
 switchport mode access
 switchport access vlan 10
 switchport protected                     ! protected ports never forward to each other
 spanning-tree portfast
 ! portfast skips the listen/learn wait; bpduguard shuts the port on a switch's BPDU
 spanning-tree bpduguard enable
interface range TenGigabitEthernet1/1/1 - 3
 description R1 uplink / BOOT01 / CACHE01  ! NOT protected: every seat reaches these
! careful: an unprotected port can reach every protected one (see tournament mode)

! ===== R1 — seats use the cache's DNS, nothing else =====
! R1 — seats may only use the cache server's DNS; block all other DNS leaving the VLAN
ip access-list extended ACL-SEATS-NODNS
 deny   udp any any eq 53
 deny   tcp any any eq 53
 deny   tcp any any eq 853
 permit ip any any
interface Te0/0/0.10
 ip access-group ACL-SEATS-NODNS in
! the cache resolver is on the seats VLAN itself, so intra-VLAN DNS still works
! ===== R1 — WAN1, masquerade, UPnP on the console VLAN only =====
! R1 — IOS has no UPnP. Open NAT for six consoles needs six public addresses:
! ask the ISP for a /28 and give each booth a 1:1 static NAT.
ip access-list standard NAT-ALL
 permit 10.30.0.0 0.0.255.255
ip nat inside source list NAT-ALL interface GigabitEthernet0/0/0 overload
ip nat inside source static 10.30.20.11 203.0.113.17      ! booth 1
ip nat inside source static 10.30.20.12 203.0.113.18      ! booth 2
! … booths 3-6
interface GigabitEthernet0/0/0
 ip nat outside
! ===== R1 — WAN2, recursive failover, and a routing table for the cache =====
! R1 — IP SLA + track instead of recursive routes, PBR instead of a routing table
ip sla 1
 icmp-echo 1.1.1.1 source-interface GigabitEthernet0/0/0
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
ip sla 2
 icmp-echo 9.9.9.9 source-interface GigabitEthernet0/0/1
ip sla schedule 2 life forever start-time now
track 2 ip sla 2 reachability
ip route 1.1.1.1 255.255.255.255 203.0.113.9
ip route 9.9.9.9 255.255.255.255 198.51.100.1     ! VDSL modem in router mode, /30 to R1
ip route 0.0.0.0 0.0.0.0 203.0.113.9 track 1
ip route 0.0.0.0 0.0.0.0 198.51.100.1 250
! cache downloads ride WAN2 while it's healthy; LAN traffic is excluded first
! read top-down, first match wins; wildcard masks inverted. Deny RFC1918, then permit internet
ip access-list extended CACHE-OUT
 deny   ip host 10.30.10.20 10.30.0.0 0.0.255.255
 permit ip host 10.30.10.20 any
route-map PBR-LAN permit 10
 match ip address CACHE-OUT
 set ip next-hop verify-availability 198.51.100.1 1 track 2
interface Vlan10
 ip policy route-map PBR-LAN
! ===== R1 — mangle and queue tree (WAN1) =====
! R1 — NBAR2 classifies by application instead of by size
class-map match-any GAME
 match protocol attribute category gaming
 match protocol attribute category voice-and-video
class-map match-any BULK
 match protocol attribute category file-sharing
policy-map WAN1-CHILD
 class GAME
  priority percent 30
 class BULK
  bandwidth remaining percent 10
 class class-default
  fair-queue
policy-map WAN1-OUT
 class class-default
  shape average 190000000
  service-policy WAN1-CHILD
interface GigabitEthernet0/0/0
 service-policy output WAN1-OUT
! the same classes, shaped at 950M, go on the LAN-facing interface's output for downloads
! check the category names on your release: show ip nbar attribute category
! ===== SW-SEATS — tournament on/off, scheduled =====
! SW-SEATS — EEM with cron timers. Because an unprotected port can reach every
! protected one, the bracket also moves to its own VLAN 70 (with its own SVI on R1).
event manager applet TOURNAMENT-ON
 event timer cron cron-entry "0 17 * * 6"
 action 1.0 cli command "enable"
 action 2.0 cli command "configure terminal"
 action 3.0 cli command "interface range GigabitEthernet1/0/1 - 10"
 action 4.0 cli command "switchport access vlan 70"
 action 5.0 cli command "no switchport protected"
 action 6.0 syslog priority warnings msg "TOURNAMENT ON: seats 1-10 in VLAN 70"
event manager applet TOURNAMENT-OFF
 event timer cron cron-entry "30 23 * * 6"
 action 1.0 cli command "enable"
 action 2.0 cli command "configure terminal"
 action 3.0 cli command "interface range GigabitEthernet1/0/1 - 10"
 action 4.0 cli command "switchport access vlan 10"
 action 5.0 cli command "switchport protected"
 action 6.0 syslog priority warnings msg "TOURNAMENT OFF: seats 1-10 isolated again"
