radius server ISE
 address ipv4 10.80.10.20 auth-port 1812 acct-port 1813

! ===== R1 — interface lists, input and forward chains =====
! R1 — Zone-Based Firewall. No zone-pair = dropped (default-deny).
zone security STAFF
zone security BERTH
zone security OT
zone security CCTV
zone security PAY
zone security WAN
! each inside VLAN joins its zone (interface VlanXX / zone-member security ...)
class-map type inspect match-any CM-GEN
 match protocol tcp
 match protocol udp
 match protocol icmp
policy-map type inspect PM-OUT
 class type inspect CM-GEN
  inspect
 class class-default
  drop
! payment reaches only the acquirer, on 443
ip access-list extended ACL-ACQ
 permit tcp any 203.0.113.0 0.0.0.255 eq 443
class-map type inspect match-all CM-ACQ
 match access-group name ACL-ACQ
policy-map type inspect PM-ACQ
 class type inspect CM-ACQ
  inspect
zone-pair security ZP-PAY-WAN source PAY destination WAN
 service-policy type inspect PM-ACQ
! staff -> OT, CCTV, WAN ; OT/CCTV/BERTH -> WAN only ; no pair = no path
zone-pair security ZP-STAFF-OT source STAFF destination OT
 service-policy type inspect PM-OUT
zone-pair security ZP-BERTH-WAN source BERTH destination WAN
 service-policy type inspect PM-OUT
! ===== R1 — fibre primary, 5G backup, failover by recursive route =====
! R1 — fibre on Gi0/0/0 (DHCP), 5G on the cellular interface as the backup
interface GigabitEthernet0/0/0
 description fibre
 ip address dhcp
 ip nat outside
ip sla 1
 icmp-echo 1.1.1.1
 frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability
ip route 0.0.0.0 0.0.0.0 dhcp 1 track 1
ip route 0.0.0.0 0.0.0.0 Cellular0/2/0 2
ip access-list standard NAT-ALL
 permit 10.80.0.0 0.0.255.255
ip nat inside source list NAT-ALL interface GigabitEthernet0/0/0 overload
! police the berth VLAN down hard when it rides the cellular backup
! ===== R1 — CAKE for the berths, priority for VoIP and card auth =====
! IOS: no CAKE. A shaper with fair-queue + an LLQ priority class for VoIP and card auth.
class-map match-any PRIORITY
 match dscp ef
 match dscp af41
policy-map WAN-CHILD
 class PRIORITY
  priority percent 20
 class class-default
  fair-queue
policy-map WAN-OUT
 class class-default
  shape average 90000000
  service-policy WAN-CHILD
interface GigabitEthernet0/0/0
 service-policy output WAN-OUT
