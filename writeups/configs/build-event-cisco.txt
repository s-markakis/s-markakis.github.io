! ===== R1 — one table per job, and rules that pick the table =====
! R1 — C8300: policy routing with tracked next hops instead of routing tables.
! IOS PBR needs fixed next hops, so each Starlink keeps its own router (not bypass mode)
! and hands R1 a stable LAN address: A = 192.168.1.1, B = 192.168.2.1, 5G = 192.168.8.1.
route-map PBR-ZONES permit 10
 match ip address CRIT-SRC                     ! staff, scanners, tills
 set ip next-hop verify-availability 192.168.1.1 1 track 1
 set ip next-hop verify-availability 192.168.8.1 2 track 3
route-map PBR-ZONES permit 20
 match ip address PROD-SRC                     ! the stream
 set ip next-hop verify-availability 192.168.2.1 1 track 2
 set ip next-hop verify-availability 192.168.1.1 2 track 1
route-map PBR-ZONES permit 30
 match ip address PUBLIC-SRC                   ! public and press
 set ip next-hop verify-availability 192.168.1.1 1 track 1
 set interface Null0                           ! A down: no other line for them
interface GigabitEthernet0/0/2.900
 ip policy route-map PBR-ZONES
! ===== R1 — one simple queue, per-phone limits inside it =====

interface GigabitEthernet0/0/2.50
